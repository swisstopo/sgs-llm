# Runtime dependencies for the geodata MCP server. Its own file, not backend's: this
# deploys as a separate container and must not inherit fastapi/pydantic, nor drag faiss
# and a 2 GB embedding model into the agent image.
#
# Pinned exactly, for the same reason backend/requirements.txt is: the image is the
# contract with the deployed infrastructure.

# MCP 2.0 renamed the client entry point and moved to snake_case model fields; it also
# supersedes FastMCP with MCPServer. Do not "upgrade" from a 1.x recipe.
mcp==2.0.0
uvicorn[standard]==0.52.0

# geosearch/swisstopo.py is written against plain httpx's API, so it is pinned here.
# httpx2 is a *different library*, and it is installed too - mcp 2.0 requires httpx2>=2.5.0
# and pulls it in transitively. Both coexist on purpose; removing either breaks something.
httpx==0.28.1

# Records and vectors: DuckDB holds the rows, the .faiss files hold their embeddings,
# and row `rid` is vector `rid` - they are written together and only correct together.
duckdb==1.5.5
faiss-cpu==1.15.0
numpy==2.5.1

# Embeddings on CPU via ONNX. fastembed is what keeps torch out of the image - the same
# model through sentence-transformers would add ~2 GB of CUDA-enabled wheels the server
# never uses. onnxruntime arrives with it and is listed only to pin it.
fastembed==0.8.0
onnxruntime==1.28.0

# Bedrock (the rerank stage) and S3.
boto3==1.43.59

# `compute`. GeoJSON only - the frontend cannot render GeoParquet
# (docs/architecture.md#data-layers-from-chat), so pyarrow is deliberately absent.
shapely==2.1.2
pyproj==3.7.2

# moto is NOT here: it stands in for S3 during local development only, and the task role
# in infra/geosearch-foundation.yaml now covers the real bucket. It lives in
# requirements-dev.txt, so the deployed image ships no test double.
